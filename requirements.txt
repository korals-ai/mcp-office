# Runtime dependencies for the workspace-tool-office sidecar image.
# Keep in sync with [project.dependencies] in pyproject.toml.

# MCP server SDK: FastMCP + Streamable HTTP transport. Exposes the
# `convert` tool the workspace agent calls over http://localhost:8090/mcp.
mcp>=1.2,<2

# The shared `toolbound` package (apps/workspace-tools/toolbound) runs every
# @mcp.tool() off the event loop, bounded by a timeout — mcp pulls anyio
# transitively, but a runtime import is declared here, not inherited.
anyio>=4.0

# HTTP client for `convert`: the file is POSTed to the platform's shared
# office service (Collabora Online, /cool/convert-to) — this image carries no
# LibreOffice of its own. mcp pulls httpx transitively, but a runtime import
# is declared here, not inherited.
httpx>=0.27

# Document AUTHORING libraries (pure-python). These back author_xlsx/docx/pptx —
# the sanctioned way the agent creates Office files, so it never hand-writes a
# corrupt one. python-pptx pulls in Pillow + lxml transitively.
# openpyxl also backs xlsx_extract_cells (structured reads, not just authoring).
openpyxl>=3.1
python-docx>=1.1
python-pptx>=0.6
